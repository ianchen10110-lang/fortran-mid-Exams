第一題　Fortran 加總程式 0 / 20

題目：指揮 AI 輸出一個能做 1 加到 N 的 Fortran 程式並完成編譯，確認可以成功執行計算並輸出結果。（20 分）

程式設計說明

  1. 以 integer(kind=8) 宣告 N、i、total，使用 64 位元整數，避免總和過大時溢位。
  2. 讀入使用者輸入的 N；若 N 小於 1，顯示提示後結束程式。
  3. 用 do 迴圈從 1 累加到 N，結果存入 total。
  4. 輸出「1 加到 N 的總和」。

AI 回答的程式（檔名 sum_to_n.f90）

program sum_to_n
    implicit none
    integer(kind=8) :: n, i, total

    print *, "請輸入正整數 N："
    read *, n

    if (n < 1) then
        print *, "N 必須是正整數"
        stop
    end if

    total = 0
    do i = 1, n
        total = total + i
    end do

    print *, "1 加到", n, "的總和 =", total
end program sum_to_n

! 程式名稱：sum_to_n，計算 1 加到 N 的總和
program sum_to_n
    ! 不允許隱含型別，所有變數都必須先宣告
    implicit none
    ! 宣告三個 64 位元整數：n 為使用者輸入、i 為迴圈計數、total 為累加總和
    integer(kind=8) :: n, i, total

    ! 顯示提示文字，請使用者輸入 N
    print *, "請輸入正整數 N："
    ! 從鍵盤讀入 N 的值
    read *, n

    ! 檢查輸入是否小於 1
    if (n < 1) then
        ! 輸入不合法時顯示錯誤訊息
        print *, "N 必須是正整數"
        ! 結束程式
        stop
    end if

    ! 將總和初始化為 0
    total = 0
    ! 迴圈：讓 i 從 1 依序執行到 n
    do i = 1, n
        ! 把目前的 i 加進總和
        total = total + i
    end do

    ! 輸出計算結果
    print *, "1 加到", n, "的總和 =", total
end program sum_to_n

program score_system
    implicit none
    integer, parameter :: n_tasks = 4, points_each = 20
    character(len=100) :: tasks(n_tasks)
    integer :: i, ans, total

    tasks(1) = "第一題：Fortran 加總程式編譯並執行成功"
    tasks(2) = "第二題：完成三次隨機驗證"
    tasks(3) = "第三題：Git 記錄原版與註解新版"
    tasks(4) = "第四題：程式碼已推送到 GitHub"

    total = 0
    do i = 1, n_tasks
        print *, trim(tasks(i))
        read *, ans
        if (ans == 1) total = total + points_each
    end do

    total = total + points_each   ! 第五題：系統成功執行
    print *, "總分 =", total, "/ 100"
end program score_system
